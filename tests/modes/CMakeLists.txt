CMAKE_MINIMUM_REQUIRED(VERSION 3.10)
PROJECT(rCFDModeTest C)

OPTION(USE_MPI "build with MPI support" ON)
INCLUDE(CheckIncludeFile)

FILE(GLOB SOURCES *.c)
INCLUDE_DIRECTORIES(${CMAKE_CURRENT_SOURCE_DIR}/../../src)

ADD_DEFINITIONS(-DRCFD_STANDALONE)

IF(MSVC)
    SET(CMAKE_RUNTIME_OUTPUT_DIRECTORY $<1:${CMAKE_SOURCE_DIR}/bin>)
    ADD_DEFINITIONS(-D_CRT_SECURE_NO_WARNINGS)
endif()

SET(SOURCES main.c ${CMAKE_CURRENT_SOURCE_DIR}/../../src/rCFD_library.h)
ADD_EXECUTABLE(rcfd_mode_test ${SOURCES})
SET_TARGET_PROPERTIES(rcfd_mode_test PROPERTIES OUTPUT_NAME rcfd_mode_test)

IF(WIN32)
    TARGET_LINK_LIBRARIES(rcfd_mode_test rcfd_transport_lib)
    ADD_CUSTOM_COMMAND(TARGET rcfd_mode_test POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy -t $<TARGET_FILE_DIR:rcfd_mode_test> $<TARGET_RUNTIME_DLLS:rcfd_mode_test>
        COMMAND_EXPAND_LISTS
    )
ELSE()
    TARGET_LINK_LIBRARIES(rcfd_mode_test rcfd_transport_lib)
ENDIF()

#=======================================

IF(USE_MPI)
  FIND_PACKAGE(MPI REQUIRED)

  IF(MPI_FOUND)
    ADD_DEFINITIONS(-DRCFD_LIB_MPI)
    INCLUDE_DIRECTORIES(${MPI_INCLUDE_PATH})
    TARGET_LINK_LIBRARIES(rcfd_mode_test ${MPI_LIBRARIES})

    IF(MPI_COMPILE_FLAGS)
      SET_TARGET_PROPERTIES(rcfd_mode_test PROPERTIES COMPILE_FLAGS "${MPI_COMPILE_FLAGS}")
    ENDIF()

    IF(MPI_LINK_FLAGS)
      SET_TARGET_PROPERTIES(rcfd_mode_test PROPERTIES LINK_FLAGS "${MPI_LINK_FLAGS}")
    ENDIF()
  ENDIF(MPI_FOUND)
ENDIF(USE_MPI)
